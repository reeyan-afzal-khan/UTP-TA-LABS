\labsection{6}{NLP Foundations: Handwriting OCR Evaluation}{sec:lab06-nlp-a}

\begin{labproject}{Evaluate OCR on real handwritten medicine names}
\textbf{Scenario.} A clinic wants to digitize handwritten medicine names before storage or search.\\
\textbf{Goal.} Build a reproducible OCR benchmark, test one controlled preprocessing change, and analyse recognition failures.

\begin{labmode}
Use \texttt{all\_datasets/lab06-ocr-handwritten-prescription.zip}. The supplied archive contains one label CSV and \textbf{89 labelled images}. Run \texttt{all\_experiments/lab06\_handwriting\_ocr\_worked.ipynb}. The Python notebook uses \texttt{pytesseract}; the Tesseract OCR executable must also be installed.
\end{labmode}

\begin{mlsteps}
  \item \textbf{Validate the archive.} Load the label CSV, verify all 89 labelled files exist, report unique labels, and display a 3-by-3 sample grid.
  \item \textbf{Run baseline OCR.} Use one fixed Tesseract configuration for every raw image and normalize predictions/labels to lowercase alphanumeric text.
  \item \textbf{Measure quality.} Report exact-match accuracy, mean normalized edit distance, character error rate (CER), and blank-prediction count.
  \item \textbf{Test one preprocessing pipeline.} Convert to grayscale, add a white border, cap very large images, and resize consistently while keeping OCR configuration unchanged.
  \item \textbf{Compare fairly.} Evaluate baseline and preprocessed images on exactly the same 89 samples and save one prediction row per image/pipeline.
  \item \textbf{Analyse failures.} Review difficult examples, distinguish blank/segmentation failures from character-confusion failures, and save an evidence figure.
\end{mlsteps}

\requiredoutput{Executed notebook, predictions CSV, metrics CSV, error-analysis CSV, sample grid, and OCR-failure figure.}
\end{labproject}

\begin{verification}
A clean run must find all 89 labelled images and reproduce the same metrics. A timeout may be used per image so one OCR failure cannot block the complete experiment.
\end{verification}

\begin{labdeliverable}
Push the notebook and generated results; create tag \texttt{lab06-final}. Report exact-match accuracy, CER, the effect of preprocessing, and the dominant failure modes. Do not claim that generic OCR is production-ready for medical handwriting.
\end{labdeliverable}
