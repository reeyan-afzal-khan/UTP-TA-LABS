\labsection{8}{Genetic Algorithms: Regression and Reproducibility}{sec:lab08-ga}

\begin{labproject}{Fit a medical-cost regression model with a Genetic Algorithm}
\textbf{Scenario.} Test whether a real-valued GA can recover useful linear-regression coefficients and compare its stochastic behaviour with an exact baseline.\\
\textbf{Goal.} Implement a GA in Python, run repeated seeds, and explain when an evolutionary optimizer is or is not justified.

\begin{labmode}
Use \texttt{all\_datasets/lab08-insurance.csv} and run \texttt{all\_experiments/lab08\_genetic\_algorithm\_regression\_worked.ipynb}. Save evidence under \texttt{results/lab08/}.
\end{labmode}

\begin{mlsteps}
  \item \textbf{Verify the evolutionary loop.} Minimize $f(x)=(x-3)^2$ and confirm the best solution approaches $x=3$.
  \item \textbf{Prepare the regression task.} Predict charges from age, BMI, children, and smoker status. Use a fixed 80/20 split with seed 42 and standardize predictors using training statistics only.
  \item \textbf{Define the chromosome.} Use five real-valued genes: intercept plus four coefficients. Fitness is training MSE. Use tournament selection, blend crossover, mutation, and elitism.
  \item \textbf{Run 10 independent seeds.} Save training RMSE, test RMSE, runtime, and convergence history for seeds 1--10.
  \item \textbf{Compare with least squares.} Fit ordinary least squares on the identical split and preprocessing, then compare test RMSE.
  \item \textbf{Run one sensitivity study.} Reduce population size from 100 to 40 and repeat the 10-seed experiment. Compare median RMSE and runtime.
\end{mlsteps}
