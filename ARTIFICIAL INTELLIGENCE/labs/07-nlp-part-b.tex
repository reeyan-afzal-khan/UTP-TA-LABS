\labsection{7}{NLP Evaluation: Airline Tweet Sentiment Classification}{sec:lab07-nlp-b}

\begin{labproject}{Build and evaluate a three-class sentiment classifier from one CSV file}
\textbf{Scenario.} An airline support team wants to classify customer tweets as negative, neutral, or positive so that complaints can be prioritized and sentiment trends can be monitored.\\
\textbf{Software.} MATLAB with the text-processing and Statistics and Machine Learning functions used in the chapter.\\
\textbf{Goal.} Build a reproducible three-class sentiment benchmark using one small CSV file, standard classification metrics, per-airline analysis, and structured error inspection.

\kagglesource{https://www.kaggle.com/datasets/chandana890/twitter-airline-sentiment-dataset}{Twitter Airline Sentiment Dataset}
\kaggleexample{https://www.kaggle.com/code/langkilde/linear-svm-classification-of-sentiment-in-tweets}{Linear SVM Classification of Sentiment in Tweets}
\repofolder{lab07\_nlp\_sentiment/}

\begin{mlsteps}
  \item \textbf{Load the CSV.} Read \texttt{Tweets.csv}. Keep at least \texttt{text}, \texttt{airline\_sentiment}, and \texttt{airline}. Verify that the sentiment labels are \texttt{negative}, \texttt{neutral}, and \texttt{positive}.
  \item \textbf{Inspect the data.} Report row count, class distribution, missing text, duplicate text, and the number of tweets per airline. Remove only rows that cannot be used and document every removal.
  \item \textbf{Create one fixed split.} With seed 42, create a stratified 70/15/15 train/validation/test split by sentiment. Save the row IDs and split labels to \texttt{results/split.csv}.
  \item \textbf{Preprocess consistently.} Convert text to lowercase and apply one documented cleaning pipeline for URLs, mentions, punctuation, and stop words. Fit the vocabulary and any feature thresholds using training data only.
  \item \textbf{Implement a baseline.} Predict the most frequent training sentiment for every validation/test tweet. This establishes a minimum benchmark for the imbalanced three-class problem.
  \item \textbf{Train Naive Bayes.} Build count or binary bag-of-words features from the training split and train a three-class Naive Bayes classifier. Use the validation split only for vocabulary or smoothing choices.
  \item \textbf{Evaluate once on the test split.} Report accuracy, macro precision, macro recall, macro F1, and a 3-by-3 confusion matrix for the baseline and Naive Bayes models.
  \item \textbf{Check airline performance.} Report Naive Bayes macro F1 separately for each airline represented in the test set. Identify the best and worst performing airline subsets and give one plausible explanation.
  \item \textbf{Analyse errors.} Review at least 12 incorrect Naive Bayes predictions. Categorize each error as neutral ambiguity, negation, sarcasm/irony, context missing, unusual vocabulary, preprocessing, or other.
\end{mlsteps}
