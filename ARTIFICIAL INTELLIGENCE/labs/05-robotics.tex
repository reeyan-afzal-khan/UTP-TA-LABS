\labsection{5}{Robotics: Sensor-to-Action Control}{sec:lab05-robotics}

\begin{labproject}{Build and evaluate an interpretable wall-following controller}
\textbf{Scenario.} A SCITOS robot must convert ultrasonic sensing into one of four movement commands.\\
\textbf{Goal.} Derive a compact threshold-rule policy from the supplied 24-sensor dataset and compare it with a stronger all-sensor reference model.

\begin{labmode}
Use \texttt{all\_datasets/lab05-sensor\_readings\_24.csv} and run \texttt{all\_experiments/lab05\_robotics\_sensor\_control\_worked.ipynb}. The file has 24 sensor columns plus one action label; it is read with \texttt{header=None}.
\end{labmode}

\begin{mlsteps}
  \item \textbf{Inspect the data.} Report class balance and sensor summary statistics. Create a fixed stratified 70/30 train-test split with seed 42.
  \item \textbf{Select four sensors using training data only.} Rank the 24 sensors by mutual information and retain the top four for the interpretable controller.
  \item \textbf{Fit an interpretable rule controller.} Tune a shallow decision-tree depth on an internal validation split. Export the final tree as explicit if--then threshold rules.
  \item \textbf{Fit a reference model.} Train a random forest on all 24 sensors using the same training partition.
  \item \textbf{Evaluate on held-out data.} Report accuracy, macro precision/recall/F1, class recall, and confusion matrices. Identify the hardest action for the rule controller.
  \item \textbf{Test robustness.} Add Gaussian sensor noise at 2\%, 5\%, and 10\% of training-set standard deviation and plot rule-controller macro F1.
\end{mlsteps}

\requiredoutput{Executed notebook, split CSV, selected-feature scores, model metrics, confusion matrices, exported rules, and noise-robustness plot.}
\end{labproject}

\begin{verification}
Feature selection, depth selection, and all tuning must use training data only. Test labels are used once for final evaluation.
\end{verification}

\begin{labdeliverable}
Push the completed Python work and create tag \texttt{lab05-final}. The README must summarize the final rule controller, compare it with the all-sensor reference, and explain why offline classification accuracy is not equivalent to safe robot performance.
\end{labdeliverable}
