\labsection{4}{Fuzzy Logic and ANFIS with Air-Quality Data}{sec:lab04-fuzzy}

\begin{labproject}{Combine expert rules with data-driven prediction}
\textbf{Scenario.} An environmental monitoring system needs an interpretable ventilation rule and a data-driven model for estimating pollution. You will build both and compare their roles.\\
\textbf{Software.} MATLAB with Fuzzy Logic Toolbox.\\
\textbf{Goal.} Build a Mamdani controller and an ANFIS regression model using the same real air-quality dataset.

\kagglesource{https://www.kaggle.com/datasets/fedesoriano/air-quality-data-set}{Air Quality Dataset --- UCI measurements}
\kaggleexample{https://www.kaggle.com/code/saharpourahmad/air-quality}{Air Quality --- exploratory analysis example}
\repofolder{lab04\_fuzzy\_anfis/}

\begin{labmode}
Download the Kaggle CSV manually and save it locally as \texttt{data/air\_quality/AirQuality.csv}. The dataset contains 9,357 hourly observations; missing measurements are coded as \texttt{-200}. Do not commit the raw CSV.
\end{labmode}

\textbf{Required run.}
\begin{lstlisting}[style=mlmatlab]
matlab -batch "run('lab04_fuzzy_anfis/src/main_lab04.m')"
\end{lstlisting}

\begin{mlsteps}
  \item \textbf{Inspect and clean.} Read \texttt{AirQuality.csv}, replace \texttt{-200} with missing values, and report the number of usable rows. Plot temperature \texttt{T} against relative humidity \texttt{RH}.
  \item \textbf{Build the fuzzy controller.} Use \texttt{T} and \texttt{RH} as inputs and ventilation demand (0--100\%) as output. Define three membership functions per input and at least five clear rules. Save membership plots and the control surface.
  \item \textbf{Check rule behaviour.} Evaluate five fixed operating points supplied in the repository template. Record the expected qualitative behaviour and the actual controller output.
  \item \textbf{Prepare the ANFIS task.} Predict \texttt{CO(GT)} from two sensor features, \texttt{PT08.S1(CO)} and \texttt{PT08.S2(NMHC)}. Use only complete rows and a fixed 70/15/15 train/validation/test split with seed 42.
  \item \textbf{Fit two models.} Train a simple linear-regression baseline and an ANFIS model using the training data. Use validation data for model selection; do not tune on the test set.
  \item \textbf{Evaluate once on test data.} Report RMSE and MAE for both models and save a predicted-versus-actual figure. Explain whether the added ANFIS complexity is justified by the test result.
\end{mlsteps}
