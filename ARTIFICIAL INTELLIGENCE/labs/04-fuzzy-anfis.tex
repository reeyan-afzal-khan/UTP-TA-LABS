\labsection{4}{Fuzzy Logic and ANFIS in MATLAB}{sec:lab04-fuzzy}

\begin{labproject}{Build, inspect, train, and save fuzzy systems}
\textbf{Coverage.} Chapter~\ref{ch:fuzzy} --- fuzzy inference and ANFIS.\\
\textbf{Source files.} \texttt{Lab04/Lab\_4\_code.m} (all examples), \texttt{ANFIS.m} (the dataset part on its own), and \texttt{diabetes.csv}.\\
\textbf{Goal.} Build a Mamdani and a Sugeno system by hand, then let ANFIS learn the parameters from data instead.
\textbf{Software.} MATLAB with Fuzzy Logic Toolbox.

\begin{labmode}
Run the examples in sequence so later models reuse the established objects and terminology.
\end{labmode}

\medskip\noindent\textbf{Part A --- Mamdani temperature controller.}\par
Create \texttt{temperatureControl}, add Temperature and FanSpeed variables, define Cold/Warm/Hot and Low/Medium/High membership functions, add the three rules, evaluate at 70, and generate the FIS surface.

\begin{lstlisting}[style=mlmatlab]
output = evalfis(fis, 70);
disp(['Fan Speed at 70 C: ', num2str(output)]);
gensurf(fis);
\end{lstlisting}

\medskip\noindent\textbf{Part B --- Sugeno controller and existing FIS.}\par
Build \texttt{speedControl} with Gaussian input membership functions and linear output functions. Then load the source example \texttt{tipper} FIS and inspect its rules.

\begin{lstlisting}[style=mlmatlab]
fis3 = readfis('tipper');
showrule(fis3);
\end{lstlisting}

\medskip\noindent\textbf{Part C --- ANFIS training and prediction.}\par
Train the single-input and two-input examples, evaluate \texttt{[12 6]}, save \texttt{trainedFIS}, and reload it.

\begin{lstlisting}[style=mlmatlab]
writefis(anfisFIS2, 'trainedFIS');
loadedFIS = readfis('trainedFIS');
\end{lstlisting}

\medskip\noindent\textbf{Part D --- Data and error analysis.}\par
Load \texttt{diabetes.csv}, normalize it, select columns 1, 2, and 9 for the first 500 rows, train ANFIS, and plot training error.

\begin{lstlisting}[style=mlmatlab]
data3 = readmatrix('diabetes.csv');
data3 = normalize(data3);
dataSubset = data3(1:500, [1 2 9]);

anfisFIS3 = anfis(dataSubset, 2);
[anfisFIS4, trainError] = anfis(dataSubset, 2);
figure;
plot(trainError);
title('ANFIS Training Error');
xlabel('Epochs');
ylabel('Error');
\end{lstlisting}

\medskip\noindent\textbf{Part E --- Smart air conditioning (AC).}\par
Use the source training table Temperature, Humidity $\to$ AC Intensity and evaluate the trained model at 37 degrees C and 65 percent humidity.

\begin{lstlisting}[style=mlmatlab]
data4 = [20 30 0.3;
         25 40 0.5;
         30 50 0.7;
         35 60 0.9;
         40 70 1.0];
anfisFIS5 = anfis(data4, 2);
result = evalfis(anfisFIS5, [37 65]);
disp(['AC Intensity: ', num2str(result)]);
\end{lstlisting}

\end{labproject}
