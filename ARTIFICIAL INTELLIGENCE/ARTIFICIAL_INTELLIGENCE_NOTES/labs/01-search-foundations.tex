\labsection{1}{BFS, DFS, and Uniform-Cost Search}{sec:lab01-search}

\begin{labproject}{Search foundations across graphs, mazes, words, and trees}
\textbf{Coverage.} Breadth-First Search, Depth-First Search, and Uniform-Cost Search.\\
\textbf{Goal.} Use the same search vocabulary across four representations and explain how the frontier rule changes behavior.

\begin{labmode}
The original Lab 1 supplies the problem instances; the reusable implementations in this chapter and Lab 3 provide the coding pattern. Work through the traces by hand before running code.
\end{labmode}

\medskip\noindent\textbf{Part A --- BFS problem set.}\par
\begin{enumerate}
  \item Find a path from \texttt{A} to \texttt{G} in the unweighted interpretation of the supplied graph.
  \item Find a path from \texttt{S} to \texttt{G} in the supplied 2D maze, where ``.'' is open and ``\#'' is a wall.
  \item Transform \texttt{hit} to \texttt{cog} using the valid word list \texttt{hit, hot, dot, dog, lot, log, cog}.
  \item Traverse the seven-node binary tree level by level.
\end{enumerate}

\medskip\noindent\textbf{Part B --- DFS problem set.}\par
Repeat the graph, maze, word-ladder, and tree tasks using DFS. For the tree, use preorder traversal.

\medskip\noindent\textbf{Part C --- UCS problem set.}\par
Use the weighted graph from the handout, then solve the weighted grid and the graph with loops. Record both the returned path and its total cost.

\begin{tryit}
For every problem, record the sequence of states removed from the frontier. Then change the neighbor order in exactly one data structure and observe which returned path changes even though reachability does not.
\end{tryit}
\end{labproject}

\begin{labdeliverable}
Submit one notebook or script containing: a BFS implementation, a DFS implementation, a UCS implementation, one successful run for each representation, and a short comparison of frontier ordering, path length, and path cost.
\end{labdeliverable}
