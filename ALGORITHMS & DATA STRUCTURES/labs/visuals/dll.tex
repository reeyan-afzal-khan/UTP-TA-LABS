\begin{adslabfigure}[A doubly linked list]
\begin{adslabgraphic}
\begin{tikzpicture}[font=\small]
\def\fw{0.78}
\foreach \v [count=\i from 0] in {10,20,30}{
  \pgfmathsetmacro{\x}{3.55*\i}
  \ifnum\i=0
    \node[utpcell,minimum width=\fw cm,minimum height=0.9cm,fill=UTPPaleGray,
          font=\scriptsize\color{UTPMuted}] (pv\i) at (\x,0) {NULL};
  \else
    \node[utpcell,minimum width=\fw cm,minimum height=0.9cm,fill=white] (pv\i) at (\x,0) {};
  \fi
  \node[utpcell,minimum width=1.0cm,minimum height=0.9cm] (dt\i) at (\x+0.89,0) {\v};
  \ifnum\i=2
    \node[utpcell,minimum width=\fw cm,minimum height=0.9cm,fill=UTPPaleGray,
          font=\scriptsize\color{UTPMuted}] (nx\i) at (\x+1.78,0) {NULL};
  \else
    \node[utpcell,minimum width=\fw cm,minimum height=0.9cm,fill=white] (nx\i) at (\x+1.78,0) {};
  \fi
}
\node[utplabel,anchor=south] at (0,0.52) {prev};
\node[utplabel,anchor=south] at (0.89,0.52) {data};
\node[utplabel,anchor=south] at (1.78,0.52) {next};
% forward links (teal, above)
\foreach \i in {0,1}{
  \pgfmathtruncatemacro{\j}{\i+1}
  \draw[utparrow] (nx\i.north) to[bend left=38] (dt\j.north);
}
% backward links (orange, below)
\foreach \i in {1,2}{
  \pgfmathtruncatemacro{\j}{\i-1}
  \draw[-{Stealth[length=2.2mm]},thick,draw=UTPOrange] (pv\i.south) to[bend left=38] (dt\j.south);
}
\node[utplabel,anchor=west,color=UTPTeal]   at (7.6,1.05) {forward: \texttt{next}};
\node[utplabel,anchor=west,color=UTPOrange] at (7.6,-1.05) {backward: \texttt{prev}};
\end{tikzpicture}
\end{adslabgraphic}
\end{adslabfigure}
