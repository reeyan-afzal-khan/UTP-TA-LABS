% =====================================================================
%  diagrams/process.tex
%  Drawing vocabulary for business processes: steps and documents,
%  flowchart symbols, databases and swimlanes, the arrows between them,
%  and the small status chips that annotate a flow.
%
%  Load it after the shared preamble:
%
%      \input{../.shared/utpnotes}
%      \utpusediagrams{process}
%
%  Every style is prefixed proc, so a flow drawn for one chapter looks
%  like a flow drawn for any other, and each figure's source stays short.
%
%      procstep, procdoc            a process step, a business document
%      procgood, procbad, procwarn  a step that went well, failed, or needs care
%      procquiet                    a step that is present but not the point
%      procterm, procdec            flowchart start/stop and decision
%      procdb, procstore            a database cylinder, a data store
%      proclane, proclanehead       a swimlane and its heading
%      proczone, proczonewarn       a tinted region grouping several steps
%      procflow, procback, procslow the normal, return and delayed arrows
%      proclink                     a dotted association, not a flow
%      proccap, procnote, prochead  annotation text
%      procchip, procchipwarn, procchipbad  status chips on a step
% =====================================================================

\tikzset{
  % --- process and document boxes ---
  procstep/.style={draw=UTPBlue,fill=UTPPaleBlue,rounded corners=2mm,
                   minimum width=2.35cm,minimum height=0.82cm,align=center,
                   font=\scriptsize\bfseries\color{UTPNavy},inner sep=1.6mm},
  procdoc/.style={draw=UTPBlue,fill=white,rounded corners=1.2mm,
                  minimum width=2.2cm,minimum height=0.8cm,align=center,
                  font=\scriptsize\color{UTPNavy},inner sep=1.4mm},
  procgood/.style={draw=UTPGreen,fill=UTPPaleTeal,rounded corners=2mm,
                   minimum width=2.35cm,minimum height=0.82cm,align=center,
                   font=\scriptsize\bfseries\color{UTPNavy},inner sep=1.6mm},
  procbad/.style={draw=UTPRed,fill=UTPPaleRed,rounded corners=2mm,
                  minimum width=2.35cm,minimum height=0.82cm,align=center,
                  font=\scriptsize\bfseries\color{UTPNavy},inner sep=1.6mm},
  procwarn/.style={draw=UTPOrange,fill=UTPPaleOrange,rounded corners=2mm,
                   minimum width=2.35cm,minimum height=0.82cm,align=center,
                   font=\scriptsize\bfseries\color{UTPNavy},inner sep=1.6mm},
  procquiet/.style={draw=UTPMuted!45,fill=UTPPaleGray,rounded corners=2mm,
                    minimum width=2.35cm,minimum height=0.82cm,align=center,
                    font=\scriptsize\color{UTPMuted},inner sep=1.6mm},
  % --- flowchart symbols ---
  procterm/.style={draw=UTPNavy,fill=UTPNavy!8,rounded rectangle,
                   minimum width=2.1cm,minimum height=0.68cm,align=center,
                   font=\scriptsize\bfseries\color{UTPNavy},inner sep=1.2mm},
  procdec/.style={draw=UTPOrange,fill=UTPPaleOrange,diamond,aspect=2.1,
                  align=center,font=\scriptsize\color{UTPNavy},inner sep=0.6mm},
  % --- containers ---
  procdb/.style={cylinder,shape border rotate=90,aspect=0.30,draw=UTPBlue,
                 fill=UTPPaleBlue,minimum height=1.25cm,minimum width=2.5cm,
                 align=center,font=\scriptsize\bfseries\color{UTPNavy}},
  procstore/.style={draw=UTPBlue,fill=UTPPaleBlue,rounded corners=2mm,align=center,
                    minimum height=1.0cm,font=\scriptsize\bfseries\color{UTPNavy},inner sep=2mm},
  proclane/.style={draw=UTPMuted!45,fill=white,inner sep=0pt},
  proclanehead/.style={font=\scriptsize\bfseries\color{UTPNavy},align=center},
  proczone/.style={draw=UTPBlue!30,fill=UTPPaleBlue!45,rounded corners=2.5mm,
                   inner sep=3mm},
  proczonewarn/.style={draw=UTPOrange!45,fill=UTPPaleOrange!55,rounded corners=2.5mm,
                       inner sep=3mm},
  % --- connectors ---
  procflow/.style={-{Stealth[length=2.2mm]},thick,draw=UTPTeal},
  procback/.style={-{Stealth[length=2.2mm]},thick,draw=UTPOrange},
  procslow/.style={-{Stealth[length=2.2mm]},thick,draw=UTPRed,dashed},
  proclink/.style={semithick,draw=UTPMuted!60,dotted},
  % --- annotation ---
  proccap/.style={font=\scriptsize\color{UTPMuted},align=center,inner sep=1pt},
  procnote/.style={font=\tiny\color{UTPMuted},align=center,inner sep=1pt},
  prochead/.style={font=\small\bfseries\color{UTPNavy},align=center},
  procchip/.style={draw=UTPTeal,fill=UTPPaleTeal,rounded corners=1mm,
                   font=\tiny\bfseries\color{UTPNavy},inner xsep=1.6mm,inner ysep=0.8mm},
  procchipwarn/.style={draw=UTPOrange,fill=UTPPaleOrange,rounded corners=1mm,
                       font=\tiny\bfseries\color{UTPNavy},inner xsep=1.6mm,inner ysep=0.8mm},
  procchipbad/.style={draw=UTPRed,fill=UTPPaleRed,rounded corners=1mm,
                      font=\tiny\bfseries\color{UTPNavy},inner xsep=1.6mm,inner ysep=0.8mm}
}
