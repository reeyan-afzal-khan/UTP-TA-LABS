! ============================================================
! Lab 9 --- HQ router, serial WAN link
! ============================================================

system-view
sysname HQ

! ---------- Find the DCE end FIRST ----------
! display interface Serial 0/0/0
!   One end is DCE and must provide clocking; the other is DTE and
!   follows it. In eNSP the DCE end depends on which side you attached
!   the cable from, and it is not obvious by inspection.
!
!   No clock rate on the DCE end = the line stays down no matter how
!   correct the IP configuration is.

interface Serial 0/0/0
 ip address 10.0.0.1 30
 clock rate 64000          ! DCE end ONLY -- remove if this end is DTE
 link-protocol hdlc        ! both ends must match
 undo shutdown
quit

interface GigabitEthernet 0/0/0
 ip address 192.168.100.1 24
 undo shutdown
quit

! ---------- OSPF across the WAN ----------
ospf 1 router-id 10.10.10.10
 area 0
  network 10.0.0.0 0.0.0.3
  network 192.168.100.0 0.0.0.255
 quit
quit

quit
save

! ---------- Then switch to PPP (BOTH ends) ----------
! system-view
! interface Serial 0/0/0
!  link-protocol ppp        ! VRP asks to confirm -- the link drops
! quit

! ---------- PAP: authenticator side ----------
! aaa
!  local-user branch password cipher Branch@123
!  local-user branch service-type ppp
! quit
! interface Serial 0/0/0
!  ppp authentication-mode pap
! quit

! ---------- CHAP: authenticator side ----------
! interface Serial 0/0/0
!  ppp authentication-mode chap
! quit

! ---------- PPP brings up a link in two phases ----------
!   LCP  negotiates the link: MTU, magic number, whether auth is required
!   NCP  then negotiates each network protocol (IPCP for IPv4)
!
! Authentication sits between them. So a link that completes LCP and never
! reaches NCP is almost always failing AUTHENTICATION, not addressing.
! Knowing which phase failed halves your troubleshooting.

! ---------- Verify ----------
! display interface Serial 0/0/0     encapsulation, clock, LCP state, DCE/DTE
! display ppp all
! display ip routing-table protocol ospf
!   Serial cost is far higher than Ethernet -- Huawei derives cost from
!   bandwidth, so a parallel Ethernet path wins automatically. Usually
!   correct, occasionally not what you intended.
