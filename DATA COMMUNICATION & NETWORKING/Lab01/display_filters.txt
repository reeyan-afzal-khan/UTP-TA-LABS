# Wireshark display filters --- Lab 1
#
# Display filters (this file) use Wireshark syntax:  tcp.port == 80
# Capture filters use BPF syntax:                    tcp port 80
# They are NOT interchangeable. Mixing them is the most common reason
# for an empty packet list.

# --- by host ---
ip.addr == 192.168.1.10             # to or from this host
!(ip.addr == 192.168.1.10)          # everything else -- see the note below
ip.src == 192.168.1.10              # from this host only
ip.dst == 192.168.1.10              # to this host only

# NOTE: never write  ip.addr != 192.168.1.10
# A packet has both a source and a destination, so that expression is true
# whenever EITHER differs -- which is nearly always. Use !(ip.addr == x).

# --- by protocol ---
dns
icmp
arp
bootp                               # DHCP: Wireshark still uses the old name
ftp                                 # control channel (readable!)
ftp-data                            # the actual file bytes

# --- TCP behaviour ---
tcp.flags.syn == 1 && tcp.flags.ack == 0   # connection attempts
tcp.flags.reset == 1                       # connections refused or torn down
tcp.analysis.retransmission                # something was resent
tcp.analysis.zero_window                   # receiver ran out of buffer

# --- HTTP ---
http.request.method == "GET"
http.response.code >= 400

# --- combining ---
ip.addr == 192.168.1.10 && tcp.port == 80
dns || icmp
